function database() {
  const native = new DatabaseSync(':memory:');
  const d = { exec: sql => native.exec(sql), prepare: sql => native.prepare(sql), transaction(fn) {
    const run = (...args) => { native.exec('BEGIN IMMEDIATE'); try { const r = fn(...args); native.exec('COMMIT'); return r; }
      catch (e) { native.exec('ROLLBACK'); throw e; } }; run.immediate = run; return run;
  } };
  return { native, d };
}
function loader(doubles) {
  const cache = new Map();
  function load(file) {
    const absolute = path.resolve(root, file);
    if (cache.has(absolute)) return cache.get(absolute).exports;
    const mod = { exports: {} }; cache.set(absolute, mod);
    const source = fs.readFileSync(absolute, 'utf8');
    const code = ts.transpileModule(source, { compilerOptions: {
      module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2022, esModuleInterop: true,
    } }).outputText;
    function localRequire(name) {
      if (name.startsWith('node:')) return require(name);
      const target = name.startsWith('.') ? path.resolve(path.dirname(absolute), name.replace(/\.ts$/, '') + '.ts') : name;
      if (Object.hasOwn(doubles, target)) return doubles[target];
      if (name.startsWith('.')) return load(target);
      throw new Error('Unexpected external dependency: ' + name);
    }
    vm.runInThisContext(`(function(require,module,exports){${code}\n})`, { filename: absolute })(localRequire, mod, mod.exports);
    return mod.exports;
  }
  return load;
}

  try {
    for (const fault of ['timeout-after-submission', 'http-500', 'http-429', 'unreadable-200', 'reported-200']) {

  } finally { globalThis.fetch = originalFetch; native.close(); }

async function control() {
  const file = path.join(root, 'scripts/test-control-dispatch.cjs');
  const source = fs.readFileSync(file, 'utf8').split("test('dispatch persists")[0];
  const mod = { exports: {} };
  const localRequire = name => name === 'typescript' ? ts : require(name);
  vm.runInThisContext(`(function(require,module,exports,__dirname){${source}\nmodule.exports = {fixture};})`, { filename: file })(localRequire, mod, mod.exports, path.dirname(file));
  for (const phase of ['dequeue', 'pre-spawn']) for (const change of ['none', 'pause', 'halt', 'ownership', 'budget', 'profile', 'account', 'partial', 'missing']) {
    const f = mod.exports.fixture();
    try {
      const id = phase + '-' + change; const node = f.seed(id, { autopilot: 1, claim: 'owned-path' });
      assert.equal(f.control.sweepAutopilot(), 1);
      let release; let pending;
      if (phase === 'pre-spawn') { f.state.prepare = () => new Promise(resolve => { release = resolve; }); pending = f.control.startQueuedNode(node); }
      if (change === 'pause') f.control.setAutopilot(id, { enabled: false });
      if (change === 'halt') f.state.halted = true;
      if (change === 'ownership') f.native.prepare("UPDATE work_nodes SET status='canceled',dispatch_state=NULL WHERE id=?").run(node);
      if (change === 'budget') f.control.setDocketBudget(id, 0);
      if (change === 'profile') f.state.profiles.set('stage-profile', undefined);
      if (change === 'account') f.native.prepare("UPDATE work_nodes SET account_id='changed-account' WHERE id=?").run(node);
      if (change === 'partial') { f.meter(node, 'reported', 0.1); f.meter(node, 'missing', null); }
      if (change === 'missing') f.meter(node, 'missing', null);
      if (phase === 'pre-spawn') { release(); await pending; } else await f.control.startQueuedNode(node);
      const expected = change === 'none' || (phase === 'dequeue' && change === 'account') ? 1 : 0;
      assert.equal(f.launches.length, expected, id);
      const activeClaims = f.native.prepare('SELECT COUNT(*) n FROM work_claims WHERE released_at IS NULL').get().n;
      if (!expected) assert.equal(activeClaims, 0);
      record('CONTROL-' + id, { launches: f.launches.length, attempted: f.attempted.length, activeClaims,
        account: f.launches[0]?.session.accountId ?? null, nodeSession: f.row(node).session_id,
        evidence: 'production Control/schema and real SQLite; preparation/session/provider/queue doubles' });

  }
  for (const scenario of ['delayed-covered-session', 'independent-goals']) {
    const f = mod.exports.fixture();
    try {
      if (scenario === 'delayed-covered-session') {
        const node = f.seed(scenario, { autopilot: 1, budget: 1 }); f.meter(node, 'previous-session', 0.2);

        record('CONTROL-' + scenario, { launches: f.launches.length, note: 'No freshness/completeness/reservation field is accepted by the meter contract.' });

        assert.equal(f.launches.length, 2);
        record('CONTROL-' + scenario, { launches: f.launches.length, note: 'Goal budgets are independent; no product global audit reservation ledger.' });

}
