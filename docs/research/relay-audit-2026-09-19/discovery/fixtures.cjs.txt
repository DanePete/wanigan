function load(rel, deps = {}, globals = {}) {
  const source = fs.readFileSync(path.join(root, rel), 'utf8');
  hashes[rel] = sha(source);
  const output = ts.transpileModule(source, { compilerOptions: {
    target: ts.ScriptTarget.ES2022, module: ts.ModuleKind.CommonJS, esModuleInterop: true,
  }}).outputText;
  const module = { exports: {} };
  const ctx = vm.createContext({ module, exports: module.exports, console, Buffer,
    URL, AbortController, setTimeout, clearTimeout, queueMicrotask,
    process: { env: {} },
    require: (id) => { if (Object.hasOwn(deps, id)) return deps[id]; throw new Error('Blocked unmocked import: ' + id); },
    ...globals,
  });
  vm.runInContext(output, ctx, { filename: rel });
  return module.exports;
}
