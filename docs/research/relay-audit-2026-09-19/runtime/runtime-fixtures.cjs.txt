// Audit-only harness. Production TypeScript is transpiled without rewriting it.
// Database, project/settings/root policy, and Electron paths are isolated fixtures.
// Git, filesystem, SQLite, shell commands and subprocesses are real.
const fs = require('node:fs');
const path = require('node:path');
const vm = require('node:vm');
const cp = require('node:child_process');
const crypto = require('node:crypto');
const { DatabaseSync } = require('node:sqlite');
const assert = require('node:assert/strict');
const { EventEmitter } = require('node:events');
const source = process.env.RUNTIME_AUDIT_SOURCE;
const base = process.env.RUNTIME_AUDIT_BASE;
if (!source || !base || !base.startsWith('/private/tmp/')) throw new Error('Set isolated source/base');
const ts = require(path.join(source, 'node_modules/typescript'));
const noop = () => {};
const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));
const sha = bytes => crypto.createHash('sha256').update(bytes).digest('hex');
const shellQuote = value => "'" + value.replaceAll("'", "'\\''") + "'";
const safeEnv = { PATH: `${path.dirname(process.execPath)}:/usr/bin:/bin:/usr/sbin:/sbin`, HOME: base, SHELL: '/bin/sh', GIT_CONFIG_NOSYSTEM: '1', GIT_CONFIG_GLOBAL: '/dev/null', WANIGAN_PROVIDER_PACKS_DIR: path.join(base, 'provider-packs'), RUNTIME_AUDIT_SOURCE: source, RUNTIME_AUDIT_BASE: base };
Object.assign(process.env, safeEnv);
function alive(pid) { try { process.kill(pid, 0); return true; } catch { return false; } }
function stop(pid, group = false) { try { process.kill(group ? -pid : pid, 'SIGKILL'); } catch {} }
function size(file) { try { return fs.statSync(file).size; } catch { return 0; } }
async function until(fn, timeout = 12000) { const end = Date.now()+timeout; while (!fn()) { if (Date.now()>end) throw new Error('fixture wait timeout'); await sleep(20); } }
function fixture(name, reopen = false, opts = {}) {
  const dir = path.join(base, name), repo = path.join(dir, 'repo');
  fs.mkdirSync(repo, { recursive: true });
  const native = new DatabaseSync(path.join(dir, 'evidence.sqlite'));
  native.exec('PRAGMA journal_mode=WAL; PRAGMA busy_timeout=10000;');
  const database = { prepare: sql => native.prepare(sql), exec: sql => native.exec(sql), transaction: fn => Object.assign((...args) => fn(...args), { immediate: (...args) => fn(...args) }) };
  const project = { id: 'fixture', name: 'Fixture', path: repo };
  const cache = new Map(), observations = [], mutable = { failSafety: false };
  const overrides = {
    './db': { db: () => database, dataDir: () => dir },
    './store': { projectById: () => project, listProjects: () => [project] },
    './settings': { getSetting: (_k, fallback) => fallback, setSetting: noop, eventRetentionDays: () => 7 },
    './roots': { assertManagedRoot: value => { assert(value === repo || value.startsWith(dir + path.sep)); } },
    './hooks': { onHookEvent: noop, pruneEvents: () => 0 },
    './halt': { halted: () => false },
    './otel': { pruneSpans: () => 0 }, './statusline': { pruneStatusObservations: () => 0 },
    './worktree-setup': { depsModeFor: () => 'link', latestWorktreeRun: () => null, projectForDirectory: () => project,
      worktreeCommands: () => ({ setup: [], teardown: [] }), runWorktreePhase: async () => { observations.push('setup'); return null; } },
  };
  function load(relative) {
    const absolute = path.resolve(source, relative);
    if (cache.has(absolute)) return cache.get(absolute).exports;
    const mod = { exports: {} }; cache.set(absolute, mod);
    const text = ts.transpileModule(fs.readFileSync(absolute, 'utf8'), { compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2022, esModuleInterop: true } }).outputText;
    function localRequire(name) {
      if (name === 'electron') return { app: { getPath: () => dir }, dialog: {} };
      if (name === './checkpoints' && absolute.endsWith('/queue.ts')) return { pruneCheckpoints: () => 0 };
      if (name === './worktrees' && absolute.endsWith('/review.ts')) return { repoRootFor: async () => repo };
      if (name === 'node:child_process' && absolute.endsWith('/worktrees.ts') && opts.cloneFailure) {
        return { ...cp, spawn: (file, args) => {
          assert.equal(file, '/bin/cp'); observations.push({ cloneFailure: opts.cloneFailure, file, args });
          const child = new EventEmitter(); child.stderr = new EventEmitter(); child.kill = noop;
          setImmediate(() => { child.stderr.emit('data', Buffer.from(opts.cloneFailure)); child.emit('close', 1, null); });
          return child;
        } };
      }
      if (name === './git' && absolute.endsWith('/checkpoints.ts')) {
        const real = load('src/main/git.ts');
        return { ...real, runGit: async (cwd, args, options) => {
          if (mutable.failSafety && args[0] === 'add') return { ok: false, out: '', err: 'audit-injected safety snapshot failure', code: 1, killed: false };
          return real.runGit(cwd, args, options);
        } };
      }
      if (Object.hasOwn(overrides, name)) return overrides[name];
      if (name.startsWith('.')) return load(path.resolve(path.dirname(absolute), name + '.ts'));
      if (name.startsWith('node:')) return require(name);
      throw new Error(`Unexpected import ${name} in ${absolute}`);
    }
    vm.runInThisContext(`(function(require,module,exports){${text}\n})`, { filename: absolute })(localRequire, mod, mod.exports);
    return mod.exports;
  }
  function git(args) {
    const result = cp.spawnSync('git', ['-C', repo, '-c', 'core.hooksPath=/dev/null', '-c', 'core.fsmonitor=false', ...args], { encoding: 'utf8', env: safeEnv });
    if (result.status !== 0) throw new Error(`git ${args[0]}: ${result.stderr}`);
    return result.stdout.trim();
  }
  if (!reopen) {
    git(['init', '--initial-branch=main']);
    fs.writeFileSync(path.join(repo, '.gitignore'), 'node_modules/\n');
    fs.writeFileSync(path.join(repo, 'tracked.txt'), 'baseline\n');
    git(['add', '.']); git(['-c', 'user.name=Fixture', '-c', 'user.email=fixture@example.invalid', 'commit', '-m', 'audit fixture']);
    native.exec('CREATE TABLE projects (id TEXT PRIMARY KEY); INSERT INTO projects VALUES (\'fixture\');');
  }
  const schema = load('src/main/modules/session-storage.ts');
  schema.migrateCheckpoints(database); schema.migrateWorktrees(database);
  load('src/main/modules/review.ts').reviewModule.migrate(database);
  native.exec(`CREATE TABLE IF NOT EXISTS queue(id TEXT PRIMARY KEY,kind TEXT,state TEXT DEFAULT 'waiting',priority INTEGER DEFAULT 100,label TEXT,payload_json TEXT,blocked_by TEXT,attempts INTEGER DEFAULT 0,next_attempt_at INTEGER,created_at INTEGER,started_at INTEGER,ended_at INTEGER,error TEXT,lease_owner TEXT,lease_expires_at INTEGER);`);
  return { dir, repo, native, load, git, observations, mutable, close: () => native.close() };
}

}

async function runWorktrees() {
  const results = [];
  for (const kind of ['private-copy', 'root-external', 'nested-external', 'hardlink-existing', 'hardlink-source', 'clone-failure', 'disk-full']) {
    const f = fixture('deps-' + kind, false, { cloneFailure: kind === 'clone-failure' ? 'fixture clone failure' : kind === 'disk-full' ? 'No space left on device' : null });
    const tree = path.join(f.dir, 'checkout'); f.git(['worktree', 'add', '-b', 'audit-private', tree]);
    f.native.prepare('INSERT INTO worktrees(path,repo_root,branch,session_id,created_at) VALUES (?,?,?,?,?)').run(tree, f.repo, 'audit-private', 'audit-private', Date.now());
    const parent = path.join(f.repo, 'node_modules'); fs.mkdirSync(parent); fs.writeFileSync(path.join(parent, 'sentinel.txt'), 'parent dependency\n');
    const dest = path.join(tree, 'node_modules');
    const external = path.join(f.dir, 'external'); fs.mkdirSync(external); fs.writeFileSync(path.join(external, 'outside.txt'), 'external dependency\n');
    if (kind === 'root-external') fs.symlinkSync(external, dest);
    else if (kind === 'nested-external') fs.symlinkSync(external, path.join(parent, 'package'));
    else if (kind === 'hardlink-existing') { fs.mkdirSync(dest); fs.linkSync(path.join(parent, 'sentinel.txt'), path.join(dest, 'shared.txt')); }
    else if (kind === 'hardlink-source') fs.linkSync(path.join(parent, 'sentinel.txt'), path.join(parent, 'linked.txt'));
    const before = sha(fs.readFileSync(path.join(parent, 'sentinel.txt'))), worktrees = f.load('src/main/worktrees.ts');
    let outcome = 'prepared', error = null, privateCopy = null, relinkPrivate = null;
    try {
      await worktrees.ensurePrivateWorktreeDependencies(tree);
      privateCopy = !fs.lstatSync(dest).isSymbolicLink();
      if (kind === 'private-copy') { fs.rmSync(dest, { recursive: true }); await worktrees.relinkWorktree(tree); relinkPrivate = !fs.lstatSync(dest).isSymbolicLink(); }
    } catch (e) { outcome = 'refused'; error = e.message; }
    const row = f.native.prepare('SELECT bootstrap_json FROM worktrees WHERE path=?').get(tree);
    results.push({ id: 'W-' + kind, outcome, error, privateCopy, relinkPrivate, parentUnchanged: before === sha(fs.readFileSync(path.join(parent, 'sentinel.txt'))), sourceLinkCount: fs.statSync(path.join(parent, 'sentinel.txt')).nlink, targetLinkCount: fs.existsSync(path.join(dest, 'sentinel.txt')) ? fs.statSync(path.join(dest, 'sentinel.txt')).nlink : null, bootstrap: JSON.parse(row.bootstrap_json || '{}'), injectedBoundary: kind === 'clone-failure' || kind === 'disk-full' ? 'cp close=1 and stderr fixture; actual volume not exhausted' : null });
    f.close();
  }
  return results;
}

  if (mode === 'writer') { fs.writeFileSync(process.argv[4], String(process.pid)); setInterval(() => fs.appendFileSync(process.argv[3], 'write\n'), 30); return; }

    const f = fixture(process.argv[3], true), review = f.load('src/main/review.ts');
    const original = cp.spawn;
    cp.spawn = (...args) => { const child = original(...args); if (args[2]?.detached) fs.writeFileSync(process.argv[4], JSON.stringify({ pid: child.pid })); return child; };
    process.on('SIGTERM', () => process.exit(0));

  }
  if (mode === 'queue-owner') {

    queue.registerRunner('headless', async () => {
      const writer = cp.spawn(process.execPath, [__filename, 'writer', path.join(f.dir, `writer-${label}.log`), path.join(f.dir, `writer-${label}.pid`)], { detached: true, stdio: 'ignore', env: safeEnv }); writer.unref();

  for (const file of ['src/main/checkpoints.ts','src/main/review.ts','src/main/queue.ts','src/main/worktrees.ts']) result.sourceHashes[file] = sha(fs.readFileSync(path.join(source,file)));

    }

    result.scenarios.push(await queueTask); result.finishedAt = new Date().toISOString(); persist();
  } catch (error) { result.error = error.stack; persist(); throw error; }

}
